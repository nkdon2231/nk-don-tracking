  // ----------------------------------------------------------
  // DATABASE
  // ----------------------------------------------------------

  function getShipments() {

    const stored = localStorage.getItem(STORAGE_KEY);

    if (!stored) {

      localStorage.setItem(
        STORAGE_KEY,
        JSON.stringify(starterShipments)
      );

      return starterShipments;

    }

    try {

      return JSON.parse(stored);

    } catch (error) {

      console.error(error);

      return [];

    }

  }


  function saveDatabase(shipments) {

    localStorage.setItem(
      STORAGE_KEY,
      JSON.stringify(shipments)
    );

  }


  // ----------------------------------------------------------
  // TRACKING ID
  // ----------------------------------------------------------

  function generateTrackingId() {

    const alphabet = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";

    function group(length) {

      let result = "";

      for (let i = 0; i < length; i++) {

        result += alphabet[
          Math.floor(Math.random() * alphabet.length)
        ];

      }

      return result;

    }

    return "NKD-" + group(4) + "-" + group(4);

  }


  // ----------------------------------------------------------
  // NORMALIZE ID
  // ----------------------------------------------------------

  function normalizeTrackingId(value) {

    return value
      .trim()
      .toUpperCase()
      .replace(/\s+/g, "");

  }


  // ----------------------------------------------------------
  // STATUS HELPERS
  // ----------------------------------------------------------

  function getLatestEvent(shipment) {

    if (!shipment.events || shipment.events.length === 0) {
      return null;
    }

    return [...shipment.events].sort(
      (a, b) =>
        new Date(b.timestamp) - new Date(a.timestamp)
    )[0];

  }


  function getStatusClass(status) {

    const normalized = status.toLowerCase();

    if (normalized === "processing") {
      return "status-processing";
    }

    if (normalized === "in transit") {
      return "status-transit";
    }

    if (normalized === "arrived") {
      return "status-arrived";
    }

    if (normalized === "delivered") {
      return "status-delivered";
    }

    if (normalized === "delayed") {
      return "status-delayed";
    }

    return "status-default";

  }

  function formatDate(date) {

    if (!date) return "—";

    return new Date(date).toLocaleDateString(
      "en-GB",
      {
        day: "2-digit",
        month: "short",
        year: "numeric"
      }
    );

  }


  function formatDateTime(date) {

    if (!date) return "—";

    return new Date(date).toLocaleString(
      "en-GB",
      {
        day: "2-digit",
        month: "short",
        year: "numeric",
        hour: "2-digit",
        minute: "2-digit"
      }
    );

  }

  function trackShipment() {

    const input =
      document.getElementById("trackingInput");

    const trackingId =
      normalizeTrackingId(input.value);

    const results =
      document.getElementById("trackingResults");

    if (!trackingId) {

      results.classList.remove("hidden");

      results.innerHTML = `
        <div class="result-card">
          <strong>Please enter a tracking ID.</strong>
        </div>
      `;

      return;

    }

    const shipments = getShipments();

    const shipment = shipments.find(
      item =>
        normalizeTrackingId(item.id) === trackingId
    );

    results.classList.remove("hidden");

    if (!shipment) {

      results.innerHTML = `
        <div class="result-card">

          <div style="font-size:35px;margin-bottom:10px;">
            ×
          </div>

          <h2>Shipment not found</h2>

          <p style="color:#666;margin-top:8px;">
            We couldn't find a shipment matching
            <strong class="code">${escapeHTML(trackingId)}</strong>.
            Please check the tracking ID and try again.
          </p>

        </div>
      `;

      return;

    }

    renderTrackingResult(shipment);

  }

  function saveShipment() {

    const trackingId =
      document.getElementById(
        "shipmentTrackingId"
      ).value || generateTrackingId();

    const service =
      document.getElementById(
        "shipmentService"
      ).value;

    const origin =
      document.getElementById(
        "shipmentOrigin"
      ).value.trim();

    const destination =
      document.getElementById(
        "shipmentDestination"
      ).value.trim();

    const delivery =
      document.getElementById(
        "shipmentDelivery"
      ).value;

    const recipient =
      document.getElementById(
        "shipmentRecipient"
      ).value.trim();

    const description =
      document.getElementById(
        "shipmentDescription"
      ).value.trim();

    const carrierReference =
      document.getElementById(
        "shipmentCarrierReference"
      ).value.trim();

    const eventDescription =
      document.getElementById(
        "shipmentEvent"
      ).value.trim();


    if (!origin || !destination) {

      alert(
        "Please enter the shipment origin and destination."
      );

      return;

    }


    const shipments = getShipments();


    if (editingId) {

      const shipment =
        shipments.find(
          item => item.id === editingId
        );

      if (!shipment) return;

      shipment.service = service;
      shipment.origin = origin;
      shipment.destination = destination;
      shipment.delivery = delivery;
      shipment.recipient = recipient;
      shipment.description = description;
      shipment.carrierReference = carrierReference;

      if (eventDescription) {

        shipment.events.push({
          status:
            getLatestEvent(shipment)?.status ||
            "Processing",

          location: origin,

          description: eventDescription,

          timestamp:
            new Date().toISOString()

        });

      }

      saveDatabase(shipments);

      alert("Shipment updated successfully.");

      cancelEdit();

      renderShipments();

      refreshEventShipmentList();

      return;

    }


    const newShipment = {

      id: normalizeTrackingId(trackingId),

      service,

      origin,

      destination,

      delivery,

      recipient,

      description,

      carrierReference,

      events: [

        {

          status: "Processing",

          location: origin,

          description:
            eventDescription ||
            "Shipment received and processed.",

          timestamp:
            new Date().toISOString()

        }

      ]

    };


    shipments.unshift(newShipment);

    saveDatabase(shipments);

    alert(
      "Shipment created.\n\nTracking ID: " +
      newShipment.id
    );

    renderShipments();

    refreshEventShipmentList();

    prepareNewShipment();

  }

  // ----------------------------------------------------------

  function deleteShipment(id) {

    const shipment =
      getShipments().find(
        item => item.id === id
      );

    if (!shipment) return;

    const confirmed =
      confirm(
        "Delete shipment " +
        shipment.id +
        "?\n\nThis cannot be undone."
      );

    if (!confirmed) return;

    const shipments =
      getShipments().filter(
        item => item.id !== id
      );

    saveDatabase(shipments);

    renderShipments();

    refreshEventShipmentList();

  }

      document.getElementById(

  }


  function addTrackingEvent() {

    const shipmentId =
      document.getElementById(
        "eventShipment"
      ).value;

    const location =
      document.getElementById(
        "eventLocation"
      ).value.trim();

    const status =
      document.getElementById(
        "eventStatus"
      ).value;

    const description =
      document.getElementById(
        "eventDescription"
      ).value.trim();


    if (!shipmentId ||
        !location ||
        !description) {

      alert(
        "Please complete the event fields."
      );

      return;

    }


    const shipments = getShipments();

    const shipment =
      shipments.find(
        item => item.id === shipmentId
      );

    if (!shipment) {

      alert("Shipment not found.");

      return;

    }


    shipment.events.push({

      status,

      location,

      description,

      timestamp:
        new Date().toISOString()

    });


    saveDatabase(shipments);

    document.getElementById(
      "eventLocation"
    ).value = "";

    document.getElementById(
      "eventDescription"
    ).value = "";

    renderShipments();

    alert("Tracking event added.");

  }

                )}

                ${escapeHTML(
                  shipment.destination || "—"
                )}

                ${escapeHTML(status)}
              </small>

            <div class="row-actions">

                class="small-btn view-btn"

  // ----------------------------------------------------------
  // SECURITY HELPERS
  // ----------------------------------------------------------

  function escapeHTML(value) {

    return String(value ?? "")
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;")
      .replace(/'/g, "&#039;");

  }


  function escapeJS(value) {

    return String(value ?? "")
      .replace(/\\/g, "\\\\")
      .replace(/'/g, "\\'");

  }
